Pengaturan Web App URL

Masukkan URL Web Application Google Apps Script yang telah Anda *deploy* untuk menghubungkan spreadsheet Anda.

Panduan Google Apps Script Backend

Ikuti 4 langkah mudah ini untuk menjadikan Google Sheets sebagai database Anda:

  1. Buat **Google Sheet** baru, lalu beri nama sheet pertamanya: Sheet1.
  2. Buat header di baris ke-1: ID, Tanggal, Tipe, Kategori, Jumlah, Keterangan.
  3. Buka menu **Ekstensi > Apps Script**, lalu hapus semua kode bawaan dan tempel kode berikut:
function doGet(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1");
  var data = sheet.getDataRange().getValues();
  var result = [];
  
  // Skip header (i = 1)
  for (var i = 1; i < data.length; i++) {
    result.push({
      id: data[i][0],
      date: formatDate(data[i][1]),
      type: data[i][2],
      category: data[i][3],
      amount: data[i][4],
      note: data[i][5]
    });
  }
  
  return ContentService.createTextOutput(JSON.stringify({ status: "success", data: result }))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Sheet1");
    var postData = JSON.parse(e.postData.contents);
    
    sheet.appendRow([
      postData.id || Date.now().toString(),
      postData.date,
      postData.type,
      postData.category,
      postData.amount,
      postData.note || ""
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({ status: "success" }))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(error) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() }))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function formatDate(d) {
  if (!d) return "";
  if (d instanceof Date) {
    var year = d.getFullYear();
    var month = ("0" + (d.getMonth() + 1)).slice(-2);
    var day = ("0" + d.getDate()).slice(-2);
    return year + "-" + month + "-" + day;
  }
  return d.toString();
}
